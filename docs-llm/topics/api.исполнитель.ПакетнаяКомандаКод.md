# Код модуля менеджера

Подраздел темы «Пример: команда пакетной печати» (см. topic: api.исполнитель.ПримерКомандаПакетнойПечати).

```bsl
#Если Сервер Или ТолстыйКлиентОбычноеПриложение Или ВнешнееСоединение Тогда

#Область Настройка

// Объект, в подменю «Печать» которого появятся команды.
Функция ОбъектПечати()
    Возврат Метаданные.Документы.РеализацияТоваровУслуг;
КонецФункции

// Макеты PrintWizard, которые печатает обработка:
// идентификатор печатной формы -> наименование макета в справочнике «Макеты».
Функция МакетыPrintWizard()
    Макеты = Новый Соответствие;
    Макеты.Вставить("Накладная", "Накладная ТОРГ-12");
    Макеты.Вставить("ТТН", "Транспортная накладная");
    Возврат Макеты;
КонецФункции

// Команды пакетной печати. Печатные формы - через запятую:
//  - макет PrintWizard - идентификатор из МакетыPrintWizard();
//  - встроенная печатная форма - "<Менеджер печати>.<Идентификатор>" из консоли команд.
Функция КомандыКомплектов()
    Команды = Новый Массив;
    
    Команды.Добавить(Новый Структура("Представление, ПечатныеФормы",
        НСтр("ru='Комплект к отгрузке'"),
        "Накладная,ТТН,Документ.РеализацияТоваровУслуг.ПФ_MXL_СчетФактура"));
    
    Команды.Добавить(Новый Структура("Представление, ПечатныеФормы",
        НСтр("ru='Накладная и ТТН'"),
        "Накладная,ТТН"));
    
    Возврат Команды;
КонецФункции

#КонецОбласти

#Область ПрограммныйИнтерфейс

Процедура ПриОпределенииНастроек(Настройки) Экспорт
    Настройки.Размещение.Добавить(ОбъектПечати());
    Настройки.ДобавитьКомандыПечати = Истина;
КонецПроцедуры

Процедура ДобавитьКомандыПечати(КомандыПечати) Экспорт
    Для Каждого ОписаниеКоманды Из КомандыКомплектов() Цикл
        КомандаПечати = КомандыПечати.Добавить();
        КомандаПечати.Идентификатор = ОписаниеКоманды.ПечатныеФормы;
        КомандаПечати.Представление = ОписаниеКоманды.Представление;
    КонецЦикла;
КонецПроцедуры

// Формирует макеты PrintWizard. Встроенные печатные формы
// БСП печатает их собственными менеджерами, сюда они не попадают.
Процедура Печать(МассивОбъектов, ПараметрыПечати, КоллекцияПечатныхФорм, ОбъектыПечати, ПараметрыВывода) Экспорт
    
    ПараметрыКоманды = Новый Структура;
    ПараметрыКоманды.Вставить("МассивОбъектов", МассивОбъектов);
    ПараметрыКоманды.Вставить("ПараметрыПечати", ПараметрыПечати);
    ПараметрыКоманды.Вставить("КоллекцияПечатныхФорм", КоллекцияПечатныхФорм);
    ПараметрыКоманды.Вставить("ОбъектыПечати", ОбъектыПечати);
    ПараметрыКоманды.Вставить("ПараметрыВывода", ПараметрыВывода);
    
    Для Каждого КлючИЗначение Из МакетыPrintWizard() Цикл
        Если УправлениеПечатью.НужноПечататьМакет(КоллекцияПечатныхФорм, КлючИЗначение.Ключ) Тогда
            ВывестиМакетPrintWizard(КлючИЗначение.Ключ, КлючИЗначение.Значение,
                МассивОбъектов, ПараметрыКоманды, КоллекцияПечатныхФорм);
        КонецЕсли;
    КонецЦикла;
    
КонецПроцедуры

#КонецОбласти

#Область СлужебныеПроцедурыИФункции

Процедура ВывестиМакетPrintWizard(Идентификатор, НаименованиеМакета, МассивОбъектов, ПараметрыКоманды, КоллекцияПечатныхФорм)
    
    СсылкаНаМакет = Справочники.pw_Макеты.НайтиПоНаименованию(НаименованиеМакета, Истина);
    Если СсылкаНаМакет.Пустая() Тогда
        ВызватьИсключение СтрШаблон(НСтр("ru='Не найден макет PrintWizard ""%1""'"), НаименованиеМакета);
    КонецЕсли;
    
    Исполнитель = Обработки.pw_Исполнитель.Создать();
    Исполнитель.ПараметрыКоманды = ПараметрыКоманды; // области печати по объектам для окна печати
    Исполнитель.Загрузить(СсылкаНаМакет);
    
    Если НЕ Исполнитель.Подготовлен() Тогда
        ВызватьИсключение Исполнитель.ТекстОшибок();
    КонецЕсли;
    
    Отказ = Ложь;
    Результат = Исполнитель.Печать(МассивОбъектов,, Отказ);
    
    Если НЕ Результат.Выполнено Тогда
        Если Результат.Отказ Тогда
            // событие ПередИнициализацией отказалось от печати - остальные формы комплекта печатаются
            ОбщегоНазначения.СообщитьПользователю(Результат.ТекстОтказа);
            Возврат;
        КонецЕсли;
        ВызватьИсключение Исполнитель.ТекстОшибок();
    КонецЕсли;
    
    Если Исполнитель.IsOfficeOpenXML Тогда
        ПечатнаяФорма = УправлениеПечатью.СведенияОПечатнойФорме(КоллекцияПечатныхФорм, Идентификатор);
        Если ПечатнаяФорма <> Неопределено Тогда
            ПечатнаяФорма.СинонимМакета = НаименованиеМакета;
            ПечатнаяФорма.ОфисныеДокументы = Результат.Документы;
        КонецЕсли;
    Иначе
        УправлениеПечатью.ВывестиТабличныйДокументВКоллекцию(КоллекцияПечатныхФорм,
            Идентификатор, НаименованиеМакета, Результат.Документы);
    КонецЕсли;
    
КонецПроцедуры

#КонецОбласти

#КонецЕсли
```

| Процедура | Назначение |
|---|---|
| `ПриОпределенииНастроек` | Сообщает БСП, у какого объекта показывать команды обработки |
| `ДобавитьКомандыПечати` | Добавляет команды из `КомандыКомплектов()`. Печатные формы через запятую делают команду пакетной |
| `Печать` | Формирует макеты PrintWizard из команды: находит макет по наименованию, формирует его исполнителем и выводит в окно печати |

См. также: api.исполнитель.ПримерКомандаПакетнойПечати
